Integração Google Sheets

Leitura automática via PubHTML e gravação bidirecional via Apps Script

O portal extrai automaticamente todas as abas e colunas desta URL publicada.

Como links /pubhtml do Google são somente leitura por segurança, cole acima a URL do seu Web App do Google Apps Script para que adições, edições e exclusões gravem direto nas abas da planilha.

Código Google Apps Script (Cole em Extensões > Apps Script) 
function doPost(e) {
  function doPost(e) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(15000);
    if (!e || !e.postData || !e.postData.contents) throw new Error('Requisição vazia.');
    var req = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(String(req.aba || ''));
    if (!sheet) throw new Error('Aba não encontrada: ' + req.aba);
    var acao = String(req.acao || '');
    var lastRow = sheet.getLastRow();
    if (acao === 'adicionar') {
      if (!Array.isArray(req.valores) || !req.valores.length) throw new Error('Valores inválidos.');
      sheet.appendRow(req.valores);
    } else if (acao === 'editar' || acao === 'excluir') {
      var idx = Number(req.rowIndex);
      if (!Number.isInteger(idx) || idx < 0) throw new Error('Índice inválido.');
      var row = idx + 2; // linha 1 = cabeçalho; índice recebido começa em zero
      if (row > lastRow) throw new Error('Registro não encontrado. Atualize a página e tente novamente.');
      if (acao === 'editar') {
        if (!Array.isArray(req.valores) || !req.valores.length) throw new Error('Valores inválidos.');
        sheet.getRange(row, 1, 1, req.valores.length).setValues([req.valores]);
      } else {
        sheet.deleteRow(row);
      }
    } else {
      throw new Error('Ação não reconhecida.');
    }
    return resposta({ ok: true, acao: acao, aba: req.aba });
  } catch (err) {
    return resposta({ ok: false, erro: String(err.message || err) });
  } finally {
    try { lock.releaseLock(); } catch (e2) {}
  }
}
